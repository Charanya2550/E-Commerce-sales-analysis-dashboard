E-Commerce sales analysis dashboard
import pandas as pd

#loading datasets
customers=pd.read_csv('customers.csv')
order_items=pd.read_csv('order_items.csv')
orders=pd.read_csv('orders.csv')
products=pd.read_csv('products.csv')

#Inspect the columns and data types
print("--CUSTOMERS--")
display(customers.head(2))
print(customers.info())
print("--ORDERS--")
display(orders.head(2))
print(orders.info())
print("--ORDER ITEMS--")
display(order_items.head(2))
print(order_items.info())
print("--PRODUCTS--")
display(products.head(2))
print(products.info())

#Merge tables into master dataframe
item_product=pd.merge(order_items, products, on='product_id', how='left')
order_customer=pd.merge(orders, customers, on='customer_id', how='left')
master_df=pd.merge(item_product, order_customer, on='order_id', how='inner')
print("Master Dataframe Shape(Rows, Columns):", master_df.shape)
display(master_df.head(5))

#Data cleaning
#convert date columns to actual datetime objects
master_df['order_date']=pd.to_datetime(master_df['order_date'])
#extract useful date features for trend analysis
master_df['Order_Year']=master_df['order_date'].dt.year
master_df['Order_Month']=master_df['order_date'].dt.month
master_df['Order_Month_Year']=master_df['order_date'].dt.to_period('M')

#feature engineering
#calculate total net revenue per line item
master_df['Total_Revenue']=master_df['quantity']*master_df['unit_price']*(master_df['discount_amount'])
#check for any remaining missing values or data type issues
print("--Cleaned master info--")
print(master_df.info())
print("\n-- Summary of total revenue--")
print(master_df['Total_Revenue'].describe())

#Exploratory data analysis
#overall business metrics
total_revenue_sum=master_df['Total_Revenue'].sum()
total_orders=master_df['order_id'].nunique()
average_order_value=total_revenue_sum/total_orders

print(f"---key performance indiactors---")
print(f"Total Revenue: ${total_revenue_sum:,.2f}")
print(f"Total Unique Orders: {total_orders:,}")
print(f"Average Order value (AOV): ${average_order_value:,.2f}")

#Top 5 product categories by revenue
print("\n--Top 5 Categories by revenue--")
top_categories=master_df.groupby('category')['Total_Revenue'].sum().reset_index()
top_categories=top_categories.sort_values(by='Total_Revenue', ascending=False).head(5)
print(top_categories)

#Monthly sales Trend
print("\n--Monthly revenue trend--")
monthly_sales=master_df.groupby('Order_Month_Year')['Total_Revenue'].sum().reset_index()
print(monthly_sales.head(10))
master_df.to_excel('ecommerce_master_cleaned.xlsx', index=False)
print("succesfully exported 'ecommerce_master_cleaned.xlsx' to your folder!")

#data visualization using python libraries
import matplotlib.pyplot as plt
import seaborn as sns

#set visual style
sns.set_theme(style="whitegrid")

#chart 1
plt.figure(figsize=(10,5))
ax1=sns.barplot(data=top_categories, x='category', y='Total_Revenue', palette='Blues_r')
plt.title('Top  product categories by total revenue', fontsize=14, fontweight='bold')
plt.xlabel('Product category', fontsize=12)
plt.ylabel('Total Revenue ($)', fontsize=12)
plt.xticks(rotation=15)
plt.tight_layout()
plt.show()

#chart 2
monthly_sales=master_df.groupby('Order_Month_Year')['Total_Revenue'].sum().reset_index()
monthly_sales['Order_Month_Year_Str']=monthly_sales['Order_Month_Year'].astype(str)

plt.figure(figsize=(12,5))
ax1=sns.lineplot(data=monthly_sales, x='Order_Month_Year_Str', y='Total_Revenue', marker='o',color='teal', linewidth=2.5)
plt.title('E-commerce Monthly sales trend', fontsize=14, fontweight='bold')
plt.xlabel('Month-Year', fontsize=12)
plt.ylabel('Total Revenue ($)', fontsize=12)
plt.xticks(rotation=45)
plt.tight_layout()
plt.show()
